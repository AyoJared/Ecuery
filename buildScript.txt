bash <<'DEPLOY'
set -euo pipefail

apt-get update
apt-get install -y curl ca-certificates python3-venv nginx

curl -fsSL https://deb.nodesource.com/setup_22.x -o /tmp/ecuery-node-setup.sh
bash /tmp/ecuery-node-setup.sh
apt-get install -y nodejs

cd /root/Ecuery/backend
python3 -m venv .venv
.venv/bin/pip install -r requirements.txt

cd /root/Ecuery/frontend
npm ci
NEXT_PUBLIC_API_MODE=live ECUERY_API_URL=http://127.0.0.1:8000 npm run build

cat > /etc/systemd/system/ecuery-api.service <<'EOF'
[Unit]
Description=Ecuery API
Wants=network-online.target
After=network-online.target

[Service]
WorkingDirectory=/root/Ecuery/backend
ExecStart=/root/Ecuery/backend/.venv/bin/uvicorn main:app --host 127.0.0.1 --port 8000
Restart=always
RestartSec=5
Environment=PYTHONUNBUFFERED=1
UMask=0077

[Install]
WantedBy=multi-user.target
EOF

cat > /etc/systemd/system/ecuery-web.service <<'EOF'
[Unit]
Description=Ecuery frontend
Wants=network-online.target ecuery-api.service
After=network-online.target ecuery-api.service

[Service]
WorkingDirectory=/root/Ecuery/frontend
Environment=NODE_ENV=production
Environment=NEXT_PUBLIC_API_MODE=live
Environment=ECUERY_API_URL=http://127.0.0.1:8000
ExecStart=/usr/bin/node node_modules/next/dist/bin/next start --hostname 127.0.0.1 --port 3000
Restart=always
RestartSec=5

[Install]
WantedBy=multi-user.target
EOF

cat > /etc/nginx/sites-available/ecuery <<'EOF'
server {
    listen 80;
    server_name 64.177.50.176;

    location / {
        proxy_pass http://127.0.0.1:3000;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_read_timeout 180s;
        proxy_buffering off;
    }
}
EOF

ln -sfn /etc/nginx/sites-available/ecuery /etc/nginx/sites-enabled/ecuery
nginx -t

systemctl daemon-reload
systemctl enable --now ecuery-api ecuery-web nginx
systemctl reload nginx

if command -v ufw >/dev/null && ufw status | grep -q 'Status: active'; then
    ufw allow 80/tcp
fi

curl --retry 12 --retry-connrefused --retry-delay 2 \
    --fail http://127.0.0.1:8000/health
curl --retry 12 --retry-connrefused --retry-delay 2 \
    --fail -I -H 'Host: 64.177.50.176' http://127.0.0.1:3000/

echo 'Deployment finished.'
DEPLOY
